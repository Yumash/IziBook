<script>
(() => {
  let words = [], index = 0, playing = false, timer = null;

  const $ = id => document.getElementById(id);
  const loader = $('loader'), reader = $('reader'), wordEl = $('word'), stage = $('stage');
  const progressFill = $('progressFill'), wordCounter = $('wordCounter'), fileInput = $('fileInput');
  const pasteArea = $('pasteArea'), startBtn = $('startBtn'), playBtn = $('playBtn'), resetBtn = $('resetBtn');
  const speedSlider = $('speed'), speedVal = $('speedVal'), wpsSlider = $('wordsPerStep'), wpsVal = $('wordsPerStepVal'), autoGroup = $('autoGroup'), status = $('status');

  // ---------- PWA: встроенный manifest ----------
  const manifest = {
    name: "RSVP Reader", short_name: "RSVP",
    start_url: ".", display: "standalone",
    background_color: "#1a1a1a", theme_color: "#1a1a1a",
    icons: [{ src: "data:image/svg+xml," + encodeURIComponent('<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><rect fill="#3a7bd5" width="100" height="100" rx="20"/><text x="50" y="68" font-size="50" text-anchor="middle" fill="white">📖</text></svg>'), sizes: "any", type: "image/svg+xml", purpose: "any" }]
  };
  const manifestBlob = new Blob([JSON.stringify(manifest)], { type: 'application/manifest+json' });
  const manifestURL = URL.createObjectURL(manifestBlob);
  const link = document.createElement('link'); link.rel = 'manifest'; link.href = manifestURL;
  document.head.appendChild(link);

  // ---------- PWA: service worker ----------
  const swCode = `
    const CACHE = 'rsvp-v1';
    self.addEventListener('install', e => { self.skipWaiting(); });
    self.addEventListener('activate', e => { e.waitUntil(clients.claim()); });
    self.addEventListener('fetch', e => {
      e.respondWith(
        caches.match(e.request).then(cached => cached || fetch(e.request).then(res => {
          if (res.ok && e.request.method === 'GET') {
            const clone = res.clone();
            caches.open(CACHE).then(c => c.put(e.request, clone));
          }
          return res;
        }).catch(() => cached))
      );
    });
  `;
  if ('serviceWorker' in navigator) {
    const swBlob = new Blob([swCode], { type: 'application/javascript' });
    navigator.serviceWorker.register(URL.createObjectURL(swBlob)).catch(() => {});
  }

  // ---------- Извлечение текста из EPUB ----------
  async function extractEpubText(file) {
    status.textContent = 'Читаю EPUB...';
    const arrayBuffer = await file.arrayBuffer();
    const book = ePub(arrayBuffer);
    await book.ready;

    // Получаем список всех секций (глав) из spine
    const spineItems = book.spine.spineItems;
    let fullText = '';

    for (const item of spineItems) {
      try {
        const doc = await item.load(book.load.bind(book));
        // doc — это Document, забираем текст
        const text = doc.body ? doc.body.textContent : doc.textContent;
        if (text) fullText += text + '\n\n';
        item.unload();
      } catch (e) {
        console.warn('Пропущена глава:', e);
      }
    }

    if (!fullText.trim()) throw new Error('EPUB не содержит извлекаемого текста');
    return fullText;
  }

  // ---------- Парсинг в слова ----------
  function parseToWords(text) {
    return text
      .replace(/\r/g, '')
      .replace(/[ \t]+/g, ' ')
      .replace(/\n{3,}/g, '\n\n')
      .split(/\s+/)
      .filter(w => w.length > 0);
  }

  // ---------- Загрузка файла ----------
  fileInput.addEventListener('change', async e => {
    const file = e.target.files[0];
    if (!file) return;
    status.textContent = '';
    try {
      let text;
      if (file.name.toLowerCase().endsWith('.epub') || file.type === 'application/epub+zip') {
        text = await extractEpubText(file);
      } else {
        text = await file.text();
      }
      startReading(text);
    } catch (err) {
      status.textContent = 'Ошибка: ' + err.message;
      console.error(err);
    }
  });

  startBtn.addEventListener('click', () => {
    const text = pasteArea.value.trim();
    if (!text) { status.textContent = 'Вставьте текст или выберите файл'; return; }
    startReading(text);
  });

  function startReading(text) {
    words = parseToWords(text);
    if (!words.length) { status.textContent = 'Не удалось найти слова'; return; }
    index = 0;
    loader.style.display = 'none';
    reader.style.display = 'flex';
    render();
  }

  // ---------- Логика RSVP (без изменений) ----------
  function getStepSize() {
    if (autoGroup.checked) {
      const w = words[index] || '';
      const len = w.replace(/[^\p{L}\p{N}]/gu, '').length;
      if (len > 9) return 1;
      if (len > 6) return 2;
      return 3;
    }
    return parseInt(wpsSlider.value, 10);
  }

  function render() {
    if (index >= words.length) {
      wordEl.textContent = '✓ Конец'; progressFill.style.width = '100%';
      wordCounter.textContent = `${words.length} / ${words.length}`; stop(); return;
    }
    const step = getStepSize();
    wordEl.textContent = words.slice(index, index + step).join(' ');
    const done = Math.min(index + step, words.length);
    wordCounter.textContent = `${done} / ${words.length}`;
    progressFill.style.width = (done / words.length * 100) + '%';
  }

  function scheduleNext() {
    if (!playing) return;
    const step = getStepSize(), spm = parseInt(speedSlider.value, 10);
    let delay = (step / spm) * 60000;
    const lastChunk = words.slice(index, index + step).join(' ');
    if (/[.!?…]$/.test(lastChunk)) delay *= 1.8;
    else if (/[,;:—-]$/.test(lastChunk)) delay *= 1.3;
    timer = setTimeout(() => {
      index += step; render();
      if (index < words.length) scheduleNext(); else stop();
    }, delay);
  }

  function play() { if (index >= words.length) index = 0; playing = true; playBtn.textContent = '⏸ Пауза'; scheduleNext(); }
  function stop() { playing = false; playBtn.textContent = '▶ Старт'; clearTimeout(timer); }

  playBtn.addEventListener('click', () => playing ? stop() : play());
  resetBtn.addEventListener('click', () => { stop(); index = 0; render(); });
  stage.addEventListener('click', () => playing ? stop() : play());

  document.addEventListener('keydown', e => {
    if (reader.style.display === 'none') return;
    if (e.code === 'Space') { e.preventDefault(); playing ? stop() : play(); }
    else if (e.code === 'ArrowRight') { e.preventDefault(); stop(); index = Math.min(index + getStepSize(), words.length - 1); render(); }
    else if (e.code === 'ArrowLeft') { e.preventDefault(); stop(); index = Math.max(0, index - getStepSize()); render(); }
  });

  speedSlider.addEventListener('input', () => { speedVal.textContent = speedSlider.value; if (playing) { clearTimeout(timer); scheduleNext(); } });
  wpsSlider.addEventListener('input', () => { wpsVal.textContent = wpsSlider.value; if (playing) { clearTimeout(timer); scheduleNext(); } render(); });
  autoGroup.addEventListener('change', () => { wpsSlider.disabled = autoGroup.checked; if (playing) { clearTimeout(timer); scheduleNext(); } render(); });
})();
</script>
